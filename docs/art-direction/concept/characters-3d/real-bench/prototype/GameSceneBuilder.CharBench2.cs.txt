        private static GameObject SpawnFigure(Transform parent, string id, float x)
        {
            const BindingFlags F = BindingFlags.Static | BindingFlags.NonPublic | BindingFlags.Public;
            GameObject fig;
            if (id == "")
            {
                var appType = typeof(Prims).Assembly.GetType("PCShopSell.Game.Runtime.FigureAppearance");
                var mi = typeof(Prims).GetMethod("Figure", F, null, new[] { typeof(Transform), typeof(string), typeof(Color), typeof(Color), typeof(float), appType }, null);
                fig = (GameObject)mi.Invoke(null, new object[] { parent, "player", ArtDirection.Clay, new Color(.95f, .8f, .65f), 1.0f, System.Enum.ToObject(appType, 1) });
            }
            else
            {
                var mi = typeof(Prims).GetMethod("FigureForPerson", F);
                fig = (GameObject)mi.Invoke(null, new object[] { parent, id, new Color(.5f, .5f, .5f), new Color(.93f, .75f, .6f), 1.0f, id, null });
                CityNpcClothing.Apply(fig, _game.Session, id);
            }
            fig.transform.localPosition = new Vector3(x, 0, 0); fig.transform.localRotation = Quaternion.Euler(0, 180, 0);
            return fig;
        }

        private static void StyleFigure(GameObject fig, float headScale)
        {
            var toon = Shader.Find("Bench/Toon"); var outline = Shader.Find("Bench/Outline");
            var outlineMat = new Material(outline);
            foreach (var head in fig.GetComponentsInChildren<Transform>(true).Where(t => t.name == "Head")) head.localScale = Vector3.one * headScale;
            foreach (var r in fig.GetComponentsInChildren<Renderer>(true).ToArray())
            {
                if (r.name == "Outline" || r.sharedMaterials.Length == 0) continue;
                var mats = r.sharedMaterials.Select(m =>
                {
                    var n = new Material(toon);
                    Color c = m != null ? (m.HasProperty("_BaseColor") ? m.GetColor("_BaseColor") : m.color) : Color.gray;
                    n.SetColor("_BaseColor", c); return n;
                }).ToArray();
                var original = r.sharedMaterials;
                r.sharedMaterials = mats;
                if (r is SkinnedMeshRenderer s && s.sharedMesh != null)
                {
                    var go = new GameObject("Outline"); go.transform.SetParent(s.transform, false);
                    var o = go.AddComponent<SkinnedMeshRenderer>();
                    o.sharedMesh = s.sharedMesh; o.bones = s.bones; o.rootBone = s.rootBone; o.localBounds = s.localBounds;
                    o.sharedMaterials = Enumerable.Repeat(outlineMat, s.sharedMesh.subMeshCount).ToArray();
                }
                else if (r is MeshRenderer mr && r.GetComponent<MeshFilter>() != null && r.GetComponent<MeshFilter>().sharedMesh != null)
                {
                    var go = new GameObject("Outline"); go.transform.SetParent(r.transform, false);
                    go.AddComponent<MeshFilter>().sharedMesh = r.GetComponent<MeshFilter>().sharedMesh;
                    go.AddComponent<MeshRenderer>().sharedMaterials = Enumerable.Repeat(outlineMat, r.GetComponent<MeshFilter>().sharedMesh.subMeshCount).ToArray();
                }
            }
        }
